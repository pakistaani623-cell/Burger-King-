"use client";

import { useEffect, useState, useMemo } from "react";
import Image from "next/image";
import { motion, AnimatePresence } from "framer-motion";
import { useStore, MenuItem } from "../lib/store";

// --- MOCK DB & DATA ---
const MENU_DB: MenuItem[] =[
  { id: "1", name: "Whopper", price: 7.99, cal: 670, category: "burger", image: "/placeholder-whopper.png" },
  { id: "2", name: "Chicken Fries", price: 4.99, cal: 420, category: "chicken", image: "/placeholder-fries.png" },
  { id: "3", name: "Oreo Shake", price: 4.49, cal: 710, category: "shake", image: "/placeholder-shake.png" },
  { id: "4", name: "Spicy Zinger", price: 6.99, cal: 550, category: "burger", image: "/placeholder-zinger.png" },
  { id: "5", name: "Iced Vanilla Coffee", price: 3.49, cal: 210, category: "beverage", image: "/placeholder-coffee.png" },
  { id: "6", name: "Lunch Share-Box", price: 14.99, cal: 1800, category: "combo", image: "/placeholder-sharebox.png" },
];

export default function ReclaimTheFlame() {
  const [mounted, setMounted] = useState(false);
  const { cart, cartTotal, upsellTriggeredItem, addToThrone, triggerUpsell, closeUpsell } = useStore();

  useEffect(() => {
    setMounted(true);
  },[]);

  // --- AI SMART MENU LOGIC ---
  // Context: Karachi, Pakistan | Time: ~2:00 PM | Temp: 29°C
  // Action: Pin cold items and lunch combos to the top.
  const smartMenu = useMemo(() => {
    const karachiHotLunchPriority = ["Iced Vanilla Coffee", "Oreo Shake", "Lunch Share-Box"];
    return[...MENU_DB].sort((a, b) => {
      const aScore = karachiHotLunchPriority.includes(a.name) ? 1 : 0;
      const bScore = karachiHotLunchPriority.includes(b.name) ? 1 : 0;
      return bScore - aScore;
    });
  },[]);

  // --- JSON-LD SCHEMA ---
  const jsonLd = {
    "@context": "https://schema.org",
    "@type": "Menu",
    name: "Burger King Reclaim the Flame 2026",
    hasMenuSection: {
      "@type": "MenuSection",
      name: "Smart Menu",
      hasMenuItem: MENU_DB.map((item) => ({
        "@type": "MenuItem",
        name: item.name,
        offers: {
          "@type": "Offer",
          price: item.price,
          priceCurrency: "USD",
        },
        nutrition: {
          "@type": "NutritionInformation",
          calories: `${item.cal} calories`,
        },
      })),
    },
  };

  if (!mounted) return <div className="min-h-screen bg-[#F5EAD6]" />; // Ghost state to prevent hydration mismatch

  return (
    <div className="relative w-full h-full">
      {/* SEO Schema Injection */}
      <script type="application/ld+json" dangerouslySetInnerHTML={{ __html: JSON.stringify(jsonLd) }} />

      {/* --- HERO COMPONENT --- */}
      <section className="relative w-full h-[400px] flex flex-col items-center justify-center p-6 bg-gradient-to-b from-[#ff8732]/20 to-transparent">
        <motion.div
          initial={{ y: -50, opacity: 0, rotateX: 45 }}
          animate={{ y: 0, opacity: 1, rotateX: 0 }}
          transition={{ type: "spring", stiffness: 100, damping: 10 }}
          className="relative z-10 w-full max-w-[300px] aspect-square drop-shadow-2xl"
        >
          {/* Ghost Loading State via background circle & Next Image priority */}
          <div className="absolute inset-0 bg-[#502314]/10 rounded-full blur-2xl animate-pulse" />
          <div className="w-full h-full bg-[#D62300] rounded-full flex items-center justify-center text-[#F5EAD6] font-bold text-3xl shadow-[0px_20px_0px_0px_#502314] border-8 border-[#502314] rotate-[-5deg]">
            3D WHOPPER
          </div>
        </motion.div>

        {/* Floating Reorder Button */}
        <motion.button
          initial={{ scale: 0, opacity: 0 }}
          animate={{ scale: 1, opacity: 1 }}
          transition={{ delay: 0.8, type: "spring" }}
          className="absolute bottom-10 z-20 bg-[#F5EAD6] text-[#D62300] border-4 border-[#D62300] px-6 py-3 rounded-full font-extrabold text-lg shadow-[4px_4px_0px_0px_#D62300] active:translate-y-1 active:shadow-none transition-all"
        >
          ⚡ Reorder My Usual
        </motion.button>
      </section>

      {/* --- AI SMART MENU COMPONENT --- */}
      <section className="px-6 py-8">
        <div className="flex justify-between items-end mb-6">
          <h2 className="text-3xl font-black uppercase tracking-tighter text-[#D62300]">
            Your Flame
          </h2>
          <span className="text-sm font-bold opacity-60">Karachi • 29°C</span>
        </div>

        <div className="grid grid-cols-2 gap-4">
          {smartMenu.map((item, idx) => (
            <motion.div
              key={item.id}
              initial={{ opacity: 0, y: 20 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ delay: idx * 0.1 }}
              className="bg-white rounded-2xl p-4 border-4 border-[#502314] shadow-[4px_4px_0px_0px_#502314] flex flex-col justify-between h-[220px]"
            >
              <div>
                <div className="text-xs font-bold text-[#FF8732] mb-1">{item.cal} Cal</div>
                <h3 className="text-lg font-black leading-tight mb-2">{item.name}</h3>
              </div>
              <div className="mt-auto">
                <div className="text-xl font-bold mb-3">${item.price}</div>
                <button
                  onClick={() => item.category === "burger" ? triggerUpsell(item) : addToThrone(item)}
                  className="w-full bg-[#D62300] text-[#F5EAD6] py-2 rounded-xl font-bold uppercase tracking-wide hover:bg-[#FF8732] transition-colors border-2 border-[#502314] shadow-[2px_2px_0px_0px_#502314] active:translate-y-[2px] active:shadow-none"
                >
                  Add to Throne
                </button>
              </div>
            </motion.div>
          ))}
        </div>
      </section>

      {/* --- UPSELL ENGINE MODAL --- */}
      <AnimatePresence>
        {upsellTriggeredItem && (
          <motion.div
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            className="fixed inset-0 z-50 bg-[#502314]/80 flex items-end justify-center sm:items-center sm:p-0"
          >
            <motion.div
              initial={{ y: "100%" }}
              animate={{ y: 0 }}
              exit={{ y: "100%" }}
              transition={{ type: "spring", bounce: 0, duration: 0.4 }}
              className="bg-[#F5EAD6] w-full max-w-md p-6 rounded-t-3xl sm:rounded-3xl border-t-8 border-[#D62300]"
            >
              <h2 className="text-3xl font-black uppercase mb-2">Upgrade Your Throne?</h2>
              <p className="font-medium mb-8">
                Make it a meal for just $3.99. Includes medium fries and a drink.
              </p>
              
              <div className="flex flex-col gap-4">
                <button
                  onClick={() => addToThrone(upsellTriggeredItem, true)}
                  className="w-full bg-[#D62300] text-white py-4 rounded-2xl font-black text-xl border-4 border-[#502314] shadow-[4px_4px_0px_0px_#502314] active:translate-y-1 active:shadow-none"
                >
                  👑 Upgrade Your Throne
                </button>
                <button
                  onClick={() => addToThrone(upsellTriggeredItem, false)}
                  className="w-full bg-transparent text-[#502314] py-4 rounded-2xl font-bold text-lg border-4 border-[#502314] hover:bg-[#502314]/10"
                >
                  Claim Your Flame (Burger Only)
                </button>
              </div>
              <button onClick={closeUpsell} className="w-full mt-4 py-2 text-sm font-bold opacity-50 uppercase">
                Cancel
              </button>
            </motion.div>
          </motion.div>
        )}
      </AnimatePresence>

      {/* --- STICKY ORDER BAR --- */}
      <AnimatePresence>
        {cart.length > 0 && (
          <motion.div
            initial={{ y: 100 }}
            animate={{ y: 0 }}
            exit={{ y: 100 }}
            className="fixed bottom-0 left-0 right-0 z-40 bg-[#502314] text-[#F5EAD6] p-4 rounded-t-3xl max-w-md mx-auto flex justify-between items-center shadow-[0px_-10px_30px_rgba(80,35,20,0.5)]"
          >
            <div>
              <div className="text-sm font-bold text-[#FF8732]">{cart.length} Items on Throne</div>
              <div className="text-2xl font-black">${cartTotal.toFixed(2)}</div>
            </div>
            
            <button className="bg-[#FF8732] text-[#502314] px-8 py-3 rounded-xl font-black text-lg uppercase tracking-wider animate-pulse border-2 border-[#F5EAD6]">
              Check Out
            </button>
          </motion.div>
        )}
      </AnimatePresence>
    </div>
  );
}
